#include <iostream>
using namespace std;

int T = 10;
int ans;
int tc, N, M;

int solution(int n, int m) {
	if (m == 0)
		return 1;
	return n * solution(n, m - 1);
}

int main() {
	for (int i = 1; i <= T; i++) {
		cin >> tc;
		cin >> N >> M;
		int ans = solution(N, M);
		cout << "#" << tc << " " <<  ans << "\n";
	}
	
}
