#define _CRT_SECURE_NO_WARNINGS
#include <bits/stdc++.h>
using namespace std;

int arr[100][100];
int T = 10;
int temp;

int main() {
	for (int t = 1; t <= T; t++) {
		int ans = 0;

		// 어차피 이건 무조건 100이다
		cin >> temp;
		for (int i = 0; i < 100; i++) {
			for (int j = 0; j < 100; j++) {
				cin >> arr[i][j];
			}
		}

		// 열마다 확인해가면서 해당열에 S가 존재하고 그전에 N이 있는지 체크하는 로직
		for (int i = 0; i < 100; i++) {
			bool flag = false;

			for (int j = 0; j < 100; j++) {
				if (arr[j][i] == 1)
					flag = true;
				if (flag && arr[j][i] == 2) {
					ans++;
					flag = false;
				}

			}
		}
		cout << "#" << t << " " << ans << "\n";
	}
}
