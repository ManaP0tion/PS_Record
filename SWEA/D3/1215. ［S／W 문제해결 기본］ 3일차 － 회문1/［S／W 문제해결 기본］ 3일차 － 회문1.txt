TC = 10

for testcase in range(1, TC+1):
    N = int(input())
    cnt = 0
    # mat 입력받자
    mat = [list(input().strip()) for _ in range(8)]

    #가로체크
    for i in range(8):
        for j in range(8-N+1):
            str = mat[i][j:j+N]
            if str == str[::-1]:
                cnt+=1

    #세로체크
    for j in range(8):
        for i in range(8-N+1):
            str = ""
            for k in range(N):
                str += mat[i+k][j]
            if str == str[::-1]:
                cnt += 1

    print(f"#{testcase} {cnt}")
